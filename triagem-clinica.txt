function visibleTotalEstimate(){
  // se pulou a personalidade, não conta os itens p*
  const skippedPersonality = answers["intro_personality"] === "later";
  return STEPS.filter(s => !(skippedPersonality && typeof s.id === "string" && s.id.startsWith("p"))).length + 1; // +1 = tela final
}

function nextIndexFrom(idx){
  let next = idx + 1;
  if (answers["intro_personality"] === "later"){
    while (next < STEPS.length && typeof STEPS[next].id === "string" && STEPS[next].id.startsWith("p")){
      next++;
    }
  }
  return next;
}

const main = document.getElementById("main");
const progressBar = document.getElementById("progress-bar");
const stepCounter = document.getElementById("step-counter");

function render(){
  if (currentIndex >= STEPS.length){
    renderFinal();
    return;
  }
  const step = STEPS[currentIndex];
  const seenSoFar = visitedPath.length;
  const total = visibleTotalEstimate();
  progressBar.style.width = Math.min(100, (seenSoFar / total) * 100) + "%";
  stepCounter.textContent = "Etapa " + seenSoFar + " de " + total;

  let html = '<div class="card">';

  if (step.type === "welcome"){
    html += `<h1 class="title">${esc(step.title)}</h1>`;
    html += `<p class="description">${esc(step.description)}</p>`;
    html += `<div class="nav-row"><button class="primary" id="btn-next">Vamos começar</button></div>`;
  } else {
    html += `<h1 class="title">${esc(step.title)}</h1>`;
    if (step.subtitle) html += `<p class="subtitle">${esc(step.subtitle)}</p>`;
    if (step.description) html += `<p class="description">${esc(step.description)}</p>`;

    if (step.type === "text_input"){
      const val = answers[step.id] || "";
      html += `<div class="field"><input type="text" id="input-field" value="${escAttr(val)}" placeholder="Digite aqui"></div>`;
    }
    if (step.type === "date_input"){
      const val = answers[step.id] || "";
      html += `<div class="field"><input type="date" id="input-field" value="${escAttr(val)}"></div>`;
    }
    if (step.type === "select"){
      html += '<div class="option-list">';
      step.options.forEach(opt => {
        const sel = answers[step.id] === opt.value ? "selected" : "";
        html += `<button type="button" class="option ${sel}" data-value="${escAttr(opt.value)}">${esc(opt.label)}</button>`;
      });
      html += '</div>';
    }
    if (step.type === "multiselect"){
      const current = answers[step.id] || [];
      if (step.maxChoices) html += `<p class="maxchoice-note">Máximo de ${step.maxChoices} escolha(s).</p>`;
      html += '<div class="option-list">';
      step.options.forEach(opt => {
        const sel = current.includes(opt.value) ? "selected" : "";
        html += `<button type="button" class="option ${sel}" data-value="${escAttr(opt.value)}">${esc(opt.label)}${opt.detail ? `<span class="detail">${esc(opt.detail)}</span>` : ""}</button>`;
      });
      html += '</div>';
    }
    if (step.type === "scale"){
      const current = answers[step.id];
      html += '<div class="scale-row">';
      for (let i = 1; i <= 6; i++){
        const sel = current === i ? "selected" : "";
        html += `<button type="button" class="scale-btn ${sel}" data-value="${i}">${i}</button>`;
      }
      html += '</div>';
      html += `<div class="scale-legend"><span>${esc(step.legendMin)}</span><span>${esc(step.legendMax)}</span></div>`;
    }
    if (step.type === "info"){
      // sem input, apenas continuar
    }

    html += `<div id="error-slot"></div>`;
    html += `<div class="nav-row">`;
    if (currentIndex > 0) html += `<button class="ghost" id="btn-back">Voltar</button>`;
    html += `<button class="primary" id="btn-next">Continuar</button>`;
    html += `</div>`;
  }

  html += '</div>';
  main.innerHTML = html;
  attachHandlers(step);
}

function attachHandlers(step){
  const nextBtn = document.getElementById("btn-next");
  const backBtn = document.getElementById("btn-back");

  if (backBtn){
    backBtn.addEventListener("click", () => {
      visitedPath.pop();
      currentIndex = visitedPath[visitedPath.length - 1];
      render();
    });
  }

  if (step.type === "select"){
    document.querySelectorAll(".option").forEach(btn => {
      btn.addEventListener("click", () => {
        answers[step.id] = btn.dataset.value;
        render();
      });
    });
  }

  if (step.type === "multiselect"){
    document.querySelectorAll(".option").forEach(btn => {
      btn.addEventListener("click", () => {
        let current = answers[step.id] || [];
        const val = btn.dataset.value;
        if (current.includes(val)){
          current = current.filter(v => v !== val);
        } else {
          if (step.maxChoices && current.length >= step.maxChoices){
            showError(`Você pode escolher no máximo ${step.maxChoices} opção(ões).`);
            return;
          }
          current = [...current, val];
        }
        answers[step.id] = current;
        render();
      });
    });
  }

  if (step.type === "scale"){
    document.querySelectorAll(".scale-btn").forEach(btn => {
      btn.addEventListener("click", () => {
        answers[step.id] = parseInt(btn.dataset.value, 10);
        render();
      });
    });
  }

  if (nextBtn){
    nextBtn.addEventListener("click", () => {
      if (!validate(step)) return;
      if (step.type === "text_input" || step.type === "date_input"){
        const el = document.getElementById("input-field");
        answers[step.id] = el.value.trim();
        if (!validate(step, true)) return;
      }
      goNext();
    });
  }
}

function validate(step, recheckInput){
  const required = typeof step.title === "string" && step.title.includes("*");
  if (!required && step.type !== "select") return true; // 'select' pode ser required sem asterisco no título em alguns casos, tratado abaixo

  if (step.type === "text_input" || step.type === "date_input"){
    const val = recheckInput ? answers[step.id] : (document.getElementById("input-field") ? document.getElementById("input-field").value.trim() : answers[step.id]);
    if (required && !val){
      showError("Este campo é obrigatório.");
      return false;
    }
  }
  if (step.type === "select" && required){
    if (!answers[step.id]){
      showError("Selecione uma opção para continuar.");
      return false;
    }
  }
  if (step.type === "multiselect" && required){
    const val = answers[step.id] || [];
    if (val.length === 0){
      showError("Selecione ao menos uma opção.");
      return false;
    }
  }
  if (step.type === "scale" && required){
    if (answers[step.id] === undefined){
      showError("Selecione um valor de 1 a 6.");
      return false;
    }
  }
  return true;
}

function showError(msg){
  const slot = document.getElementById("error-slot");
  if (slot) slot.innerHTML = `<p class="error-msg">${esc(msg)}</p>`;
}

function goNext(){
  const next = nextIndexFrom(currentIndex);
  currentIndex = next;
  visitedPath.push(currentIndex);
  render();
}

function esc(s){
  if (s === undefined || s === null) return "";
  return String(s).replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;");
}
function escAttr(s){
  return esc(s).replace(/"/g,"&quot;");
}

    const statusEl = document.getElementById("email-status");
    if (!statusEl) return;
    if (ok){
      statusEl.textContent = "Relatório enviado automaticamente por e-mail.";
    } else {
      statusEl.textContent = "Não foi possível enviar o e-mail automaticamente — use o WhatsApp ou o PDF acima para garantir que o relatório chegue.";
    }
  });
}
